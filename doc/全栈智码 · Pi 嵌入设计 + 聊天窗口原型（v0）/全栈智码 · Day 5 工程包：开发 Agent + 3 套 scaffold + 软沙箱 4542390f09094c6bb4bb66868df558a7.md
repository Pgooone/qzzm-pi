## 一、设计重点

### 1.1 noTools 按角色开启

Day 1 hardening 里已加 `noTools: role === "dev" ? undefined : "builtin"`，Day 5 **不需要再改** createSession。dev 角色启动时 Pi 内置工具（read、write、edit、bash、grep、glob）会自动可用。

### 1.2 软沙箱 = workspace 隔离 + bash 命令白名单

Pi 的 bash 工具默认能跑任何命令。Day 5 提供软沙箱：

- workspace 隔离：workspace dir = `state/projects/<id>/workspace/`（Day 1 已隔离。Pi cwd 设为该路径）。
- **命令拦截**：覆盖 Pi 内置 bash 为自定义 `safe_bash`，仅放行白名单（`ls cd cat mkdir touch echo node npm pnpm pnpm install pnpm build pnpm dev`）。
- 危险名单默认拒：`rm -rf` `sudo` `curl http*` `wget` `chmod 777` `git push` `dd` `mkfs`。
- 仅在 dev 角色下启用软沙箱。其他角色本就走 customTools、报不受影响。

### 1.3 3 套 scaffold 预置在仓库

clone 选项透过 `pick_scaffold` 工具调用，代码复制到 workspace：

| 名字 | 适用 | 包含 |
| --- | --- | --- |
| react-express | 中型全栈项目 | Vite + React 18 + TS 前端、Express + TS 后端、monorepo |
| nextjs | SSR/SEO 需求 | Next.js 14 + App Router + Tailwind + Server Actions |
| vue-express | Vue 服阶品牌 | Vite + Vue 3 + TS 前端、Express + TS 后端 |

## 四、软沙箱 + safe_bash 工具

```tsx
// apps/gateway/src/services/sandbox.ts
export const ALLOW_PREFIXES = [
  "ls", "cd", "pwd", "cat", "mkdir", "touch", "echo",
  "node", "npm", "pnpm", "npx",
  "git status", "git diff", "git log",
]
export const DENY_SUBSTRINGS = [
  "rm -rf", "rm /", "sudo ", "curl http", "wget ", "chmod 777",
  "git push", "dd if=", "mkfs", "shutdown", "reboot",
  "> /etc/", ">> /etc/",
]

export function isAllowed(cmd: string): { ok: boolean; reason?: string } {
  const c = cmd.trim()
  for (const d of DENY_SUBSTRINGS) if (c.includes(d)) return { ok: false, reason: `含黑名单片段: ${d}` }
  if (c.includes("&&") || c.includes(";") || c.includes("|")) {
    // 多命令拆检查
    for (const sub of c.split(/&&|;|\|/)) {
      const r = isAllowed(sub)
      if (!r.ok) return r
    }
    return { ok: true }
  }
  if (ALLOW_PREFIXES.some((p) => c.startsWith(p))) return { ok: true }
  return { ok: false, reason: `不在白名单: ${c.split(" ")[0]}` }
}
```

```tsx
// apps/gateway/src/pi/tools/devTools.ts
import { defineTool } from "@earendil-works/pi-coding-agent"
import { Type } from "@sinclair/typebox"
import { spawn } from "node:child_process"
import path from "node:path"
import { isAllowed } from "../../services/sandbox.js"
import { scaffoldService } from "../../services/scaffoldService.js"
import { artifactStore } from "../../store/artifactStore.js"

const STATE_DIR = process.env.STATE_DIR ?? "./state"

export function buildDevTools(projectId: string) {
  const cwd = path.resolve(STATE_DIR, "projects", projectId, "workspace")

  const safeBash = defineTool({
    name: "safe_bash",
    label: "安全 bash",
    description: "在 workspace 里跑命令。仅白名单。不要用 cd 跳出 workspace。",
    parameters: Type.Object({
      cmd: Type.String(),
      timeoutMs: Type.Optional(Type.Integer({ minimum: 1000, maximum: 600_000 })),
    }, { additionalProperties: false }),
    execute: async (_id, params, signal) => {
      const check = isAllowed(params.cmd)
      if (!check.ok) {
        return { content: [{ type: "text", text: `拒绝执行: ${check.reason}` }], details: {}, isError: true }
      }
      return await new Promise((resolve) => {
        const child = spawn("bash", ["-lc", params.cmd], { cwd })
        let out = ""; let err = ""
        const t = setTimeout(() => child.kill("SIGKILL"), params.timeoutMs ?? 60_000)
        signal?.addEventListener("abort", () => child.kill("SIGKILL"))
        child.stdout.on("data", (d) => out += d.toString())
        child.stderr.on("data", (d) => err += d.toString())
        child.on("close", (code) => {
          clearTimeout(t)
          resolve({
            content: [{ type: "text", text: `[exit ${code}]\n${out}${err ? "\n--stderr--\n" + err : ""}`.slice(0, 8000) }],
            details: { code, out, err },
            isError: code !== 0,
          })
        })
      })
    },
  })

  const pickScaffold = defineTool({
    name: "pick_scaffold",
    label: "选脚手架",
    description: "选择并展开一套脚手架到 workspace，清空原有文件。",
    parameters: Type.Object({
      template: Type.Union([Type.Literal("react-express"), Type.Literal("nextjs"), Type.Literal("vue-express")]),
      projectName: Type.String({ description: "项目名，用于 package.json" }),
    }, { additionalProperties: false }),
    execute: async (_id, params) => {
      const result = await scaffoldService.expand(cwd, params.template, params.projectName)
      return {
        content: [{ type: "text", text: `脚手架 ${params.template} 已展开。生成 ${result.fileCount} 个文件。可以跑 pnpm install 了。` }],
        details: result,
      }
    },
  })

  const readDesign = defineTool({
    name: "read_design",
    label: "读设计",
    description: "读设计文档。",
    parameters: Type.Object({}, { additionalProperties: false }),
    execute: async () => {
      try {
        const c = await artifactStore.load(projectId, "design.md", "final").catch(() => artifactStore.load(projectId, "design.md"))
        return { content: [{ type: "text", text: c }], details: {} }
      } catch (e: any) {
        return { content: [{ type: "text", text: `读不到 design.md：${e?.message}` }], details: {}, isError: true }
      }
    },
  })

  const packageZip = defineTool({
    name: "package_zip",
    label: "打包 zip",
    description: "将整个 workspace 打包为 zip 并作为 artifact。排除 node_modules 和 .git。",
    parameters: Type.Object({ filename: Type.String() }, { additionalProperties: false }),
    execute: async (_id, params) => {
      const info = await scaffoldService.zip(cwd, projectId, params.filename)
      return { content: [{ type: "text", text: `已生成 ${params.filename}，大小 ${info.size} bytes。` }], details: info }
    },
  })

  return [safeBash, pickScaffold, readDesign, packageZip]
}
```

## 六、scaffoldService.ts 核心逻辑

```tsx
import { promises as fs } from "node:fs"
import path from "node:path"
import { spawn } from "node:child_process"
import { artifactStore } from "../store/artifactStore.js"

const SCAFFOLD_ROOT = path.resolve(__dirname, "../../scaffolds")

export const scaffoldService = {
  async expand(targetCwd: string, template: string, projectName: string) {
    // 清空 workspace（保留 .versions 如果不在 workspace、实际不在）
    await fs.rm(targetCwd, { recursive: true, force: true })
    await fs.mkdir(targetCwd, { recursive: true })
    const src = path.join(SCAFFOLD_ROOT, template)
    let count = 0
    const walk = async (dir: string, rel: string) => {
      const items = await fs.readdir(dir, { withFileTypes: true })
      for (const i of items) {
        const s = path.join(dir, i.name)
        const d = path.join(targetCwd, rel, i.name)
        if (i.isDirectory()) {
          await fs.mkdir(d, { recursive: true })
          await walk(s, path.join(rel, i.name))
        } else {
          let content = await fs.readFile(s, "utf-8")
          // 变量替换
          content = content.replace(/__PROJECT_NAME__/g, projectName)
          await fs.writeFile(d, content)
          count++
        }
      }
    }
    await walk(src, "")
    return { template, projectName, fileCount: count }
  },

  async zip(srcCwd: string, projectId: string, filename: string) {
    const outDir = path.join(process.env.STATE_DIR ?? "./state", "projects", projectId, "artifacts")
    await fs.mkdir(outDir, { recursive: true })
    const outFile = path.join(outDir, filename)
    await new Promise<void>((resolve, reject) => {
      const z = spawn("zip", ["-r", outFile, ".", "-x", "node_modules/*", ".git/*", "dist/*"], { cwd: srcCwd })
      z.on("close", (c) => c === 0 ? resolve() : reject(new Error(`zip exit ${c}`)))
    })
    const stat = await fs.stat(outFile)
    return { name: filename, size: stat.size }
  },
}
```
