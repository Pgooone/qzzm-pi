## 一、设计 Agent 职责

读一版已 final 的 PRD（或最新草稿）后，输出两件：

1. **`design.md`** 业务设计文档：信息架构 · 核心流程 · 关键页面描述 · 状态/路由/服务划分 · 数据模型草图 · API 列表 · 设计决策 · 风险
2. **`ui-prototype.html`** 高保真原型：单文件 HTML，Tailwind CDN + lucide CDN + Alpine.js CDN（仅需轻交互），内嵌 mock data，**双击即开即看**。所有关键页面放在同一个 HTML 里用顶部 tab 切换。

## 四、`designTools.ts`

```tsx
import { defineTool } from "@earendil-works/pi-coding-agent"
import { Type } from "@sinclair/typebox"
import { artifactStore } from "../../store/artifactStore.js"

export function buildDesignTools(projectId: string) {
  const writeDesign = defineTool({
    name: "write_design",
    label: "写入设计产出",
    description: "写入 design.md 或 ui-prototype.html。",
    parameters: Type.Object({
      filename: Type.Union([Type.Literal("design.md"), Type.Literal("ui-prototype.html")]),
      content: Type.String({ description: "完整上下文，不是 patch" }),
      version: Type.Optional(Type.String()),
    }, { additionalProperties: false }),
    execute: async (_id, params) => {
      const info = await artifactStore.save(projectId, params.filename, params.version ?? "v1", params.content)
      return {
        content: [{ type: "text", text: `已写入 ${params.filename} 版本 ${info.version}，大小 ${info.size} bytes。` }],
        details: { name: info.name, version: info.version },
      }
    },
  })

  const readPrd = defineTool({
    name: "read_prd_for_design",
    label: "读 PRD",
    description: "读取 PRD 产出供设计参考。优先 final 版本，取不到则读 latest。",
    parameters: Type.Object({
      version: Type.Optional(Type.String({ description: "final | v1 | v2 | latest" })),
    }, { additionalProperties: false }),
    execute: async (_id, params) => {
      const tryOrder = params.version ? [params.version] : ["final", "latest"]
      for (const v of tryOrder) {
        try {
          const c = await artifactStore.load(projectId, "prd.md", v === "latest" ? undefined : v)
          return { content: [{ type: "text", text: c }], details: { version: v } }
        } catch {}
      }
      return { content: [{ type: "text", text: "未找到 PRD。建议先回需求 Agent。" }], details: {}, isError: true }
    },
  })

  return [writeDesign, readPrd]
}
```

同时修 `tools.ts` 中 `toolsForRole`：

```tsx
case "design": return buildDesignTools(projectId)
```

并修 `prdTools.ts` 的 `readPrd` 加 `version: Type.Optional(Type.String())`。
